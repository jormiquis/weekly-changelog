Decisions
AISession pattern with aggregate state

Chosen over naive per-tab tracking

export class TrackActiveTabUseCase {
  constructor(
    private repo: TimeRepository,
    private clock: Clock,
    private sessions: SessionStore,
  ) {}

  async onActiveTabChanged(info: ActiveTabInfo): Promise<void> {
    const now = this.clock.now();
    const current = await this.sessions.get();

    if (current && current.groupKey === info.groupKey) {
      if (current.groupTitle !== info.groupTitle) {
        await this.sessions.set({ ...current, groupTitle: info.groupTitle });
      }
      return;
    }

    if (current) {
      await this.flush(now);
    }
    await this.sessions.set(
      new ActiveSession(info.groupKey, info.groupTitle, now)
    );
  }

AIProvider pattern for session storage

Chosen over direct chrome.storage.local calls

export class ChromeSessionStore implements SessionStore {
  async get(): Promise<ActiveSession | null> {
    const { session } = await chrome.storage.session.get('session');
    return session ? ActiveSession.fromPlain(session) : null;
  }

  async set(session: ActiveSession): Promise<void> {
    await chrome.storage.session.set({ session });
  }
}
